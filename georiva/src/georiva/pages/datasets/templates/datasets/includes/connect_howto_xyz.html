{% trans "Use these tiles in a web map" %}

{% blocktrans %}The XYZ URL serves PNG map tiles in the standard web-map grid. It needs a time to draw, so a client has to know which times exist.{% endblocktrans %}

{{ origin }}{{ url }}

{% trans "Parameters" %}

{% if has_runs %} {% endif %} {% if private %} {% endif %}
{z}/{x}/{y} {% trans "Tile position. Your map library fills these in." %}
time {% blocktrans %}Required. ISO 8601 UTC, such as 2026-03-23T12:00:00Z. It must equal the time of an existing item exactly; otherwise the tile is a 404.{% endblocktrans %}
reftime {% blocktrans %}Required for this collection, whose data comes in runs. The run a time belongs to, in the same format. A valid time belongs to one run, so send a run's reference time together with one of its valid times.{% endblocktrans %}
style {% trans "Optional. A style name; leave it out for the variable's default." %}
api_key {% trans "Private collection: replace YOUR_KEY with a key from your account." %}

{% trans "Which times exist?" %}

{% blocktrans %}Read them from the EDR collection document. It lists every time in one response, with no paging.{% endblocktrans %}

{{ origin }}{{ edr_url }}

{% if has_runs %} {% blocktrans %}x-georiva.runs is a list of runs, newest first. Each has a reference_time and its valid_times. Send a run's reference_time as reftime and one of its valid_times as time. (extent.temporal.values holds only the newest run's times.){% endblocktrans %} {% else %} {% blocktrans %}extent.temporal.values is the list, oldest first. Each value is a valid time.{% endblocktrans %} {% endif %}

{% blocktrans %}The list covers the whole collection, not only this variable. A time this variable has no data for returns 404, so skip it. Timestamps end in +00:00; send them as Z, because a raw + in a query string is read as a space.{% endblocktrans %}

{% trans "Example" %}

// Needs a <div id="map"> and a <select id="time">.
const map = L.map("map").setView([0, 20], 4);

const EDR = "{{ origin }}{{ edr_url }}";
const TILES = "{{ origin }}{{ url }}";
const iso = (t) => t.replace("+00:00", "Z");

fetch(EDR)
  .then((r) => r.json())
  .then((doc) => {
    {% if has_runs %}// One entry per (run, valid time) pair.
    const steps = doc["x-georiva"].runs.flatMap((run) =>
      run.valid_times.map((t) => ({ time: t, reftime: run.reference_time }))
    );{% else %}const steps = doc.extent.temporal.values.map((t) => ({ time: t }));{% endif %}
    const select = document.getElementById("time");
    steps.forEach((s, i) => select.add(new Option({% if has_runs %}`${s.reftime} → ${s.time}`{% else %}s.time{% endif %}, i)));

    // Extra options fill the {Time}{% if has_runs %} and {Reftime}{% endif %} placeholders.
    const pick = (i) => ({
      Time: iso(steps[i].time),{% if has_runs %}
      Reftime: iso(steps[i].reftime),{% endif %}
    });
    const layer = L.tileLayer(TILES, pick(0)).addTo(map);

    select.onchange = () => {
      Object.assign(layer.options, pick(select.value));
      layer.redraw();
    };
  });

Leaflet TileLayer 

// Needs `map` (a maplibregl.Map) and a <select id="time">.
const EDR = "{{ origin }}{{ edr_url }}";
const TILES = "{{ origin }}{{ url }}";

const iso = (t) => t.replace("+00:00", "Z");
const tilesFor = (s) =>
  TILES.replace("{Time}", iso(s.time)){% if has_runs %}
       .replace("{Reftime}", iso(s.reftime)){% endif %};

fetch(EDR)
  .then((r) => r.json())
  .then((doc) => {
    {% if has_runs %}// One entry per (run, valid time) pair.
    const steps = doc["x-georiva"].runs.flatMap((run) =>
      run.valid_times.map((t) => ({ time: t, reftime: run.reference_time }))
    );{% else %}const steps = doc.extent.temporal.values.map((t) => ({ time: t }));{% endif %}
    const select = document.getElementById("time");
    steps.forEach((s, i) => select.add(new Option({% if has_runs %}`${s.reftime} → ${s.time}`{% else %}s.time{% endif %}, i)));

    map.on("load", () => {
      map.addSource("data", {
        type: "raster",
        tiles: [tilesFor(steps[0])],
        tileSize: 256,
      });
      map.addLayer({ id: "data", type: "raster", source: "data" });
    });

    // A raster source has no per-request parameters: give it a new URL.
    select.onchange = () =>
      map.getSource("data").setTiles([tilesFor(steps[select.value])]);
  });
