{% trans "Example" %}

from pystac_client import Client

client = Client.open("{{ origin }}{{ root_url }}"{% if private %},
                     parameters={"api_key": "YOUR_KEY"}{% endif %})
collection = client.get_collection("{{ collection_id }}")

items = collection.get_items(datetime="2026-01-01/2026-03-31")
for item in items:
    print(item.id, item.datetime)
